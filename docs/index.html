Quick Start

Add kasi-jb to your Rust project or scaffold a full service with kasi-cli: 

cargo add kasi-jb
use kasi_jb::prelude::*;

#[endpoint(GET, "/hello/{name}")]
async fn greet(Path(name): Path<String>) -> String {
    format!("Hello, {}!", name)
}

#[main]
async fn main() -> Result<(), Box<dyn std::error::Error + Send + Sync>> {
    let app = Application::builder(Config { host: "127.0.0.1".into(), port: 8080 })
        .with_observability()
        .middleware(TracingMiddleware::new())
        .route(greet)
        .build()
        .await?;

    app.serve().await?;
    Ok(())
}

Declarative Routing & Validation

Enforce regex path constraints and struct validation rules at the edge before handlers allocate resources: 

#[derive(Deserialize, Validate, Schema)]
struct CreateUser {
    #[validate(min_len = 3, regex = "^[a-z0-9_]+$")]
    username: String,
    #[validate(email)]
    email: String,
    #[validate(range(min = 18, max = 120))]
    age: u8,
}

#[endpoint(POST, "/users/{org:[a-z]+}")]
async fn create_user(
    Path(org): Path<String>,
    ValidatedJson(payload): ValidatedJson<CreateUser>,
) -> JsonValue {
    json!({ "org": org, "user": payload.username })
}

Automatic OpenAPI 3.1 & Swagger UI

Derive Schema on request/response structs. Kasi automatically generates /openapi.json and hosts an interactive Swagger UI at /docs. 

#[derive(Serialize, IntoResponse, Schema)]
struct UserResponse {
    id: uuid::Uuid,
    username: String,
}

#[endpoint(GET, "/users/{id}")]
#[response(status = 200, description = "User profile", json = UserResponse)]
async fn get_user(Path(id): Path<uuid::Uuid>) -> UserResponse {
    UserResponse { id, username: "justice".into() }
}

Security & Authentication

Protect endpoints with OpenID Connect JWKS discovery, static API keys, or Bearer tokens: 

// 1. OIDC Role & Scope Guard
#[endpoint(DELETE, "/admin/records/{id}")]
#[security(RequireRole = "Administrator", OAuth2 = ["records:delete"])]
async fn purge_record(auth: AuthContext, Path(id): Path<u64>) -> &'static str {
    "Deleted"
}

// 2. Service-to-Service API Key Middleware
let webhooks = Module::new("/webhooks")
    .wrap(from_fn(|req, next| async move { Ok(next.run(req).await) }));

Production Middleware Suite

let app = Application::builder(Config { host: "0.0.0.0".into(), port: 8080 })
    .with_observability() // Mounts GET /healthz & GET /metrics (Prometheus)
    .middleware(TracingMiddleware::json())
    .middleware(TimeoutMiddleware::new(std::time::Duration::from_secs(10)))
    .middleware(RateLimitMiddleware::new(100, std::time::Duration::from_secs(60)))
    .middleware(CompressionMiddleware::new())
    .middleware(CorsMiddleware::permissive())
    .serve_dir("/static", "public")
    .build()
    .await?;
